💬 Let's Build a Python Chatbot: ChatBuddy

print("Hi! I'm ChatBuddy, your AI friend. Let's chat!")
print("Type 'bye' to exit.\\n")

while True:
    user_input = input("You: ").lower()

    if "hello" in user_input or "hi" in user_input:
        print("ChatBuddy: Hello there! 😊")
    elif "how are you" in user_input:
        print("ChatBuddy: I'm great! Thanks for asking. How about you?")
    elif "what is your name" in user_input:
        print("ChatBuddy: My name is ChatBuddy! 🧠")
    elif "bye" in user_input:
        print("ChatBuddy: Bye! It was nice talking to you! 👋")
        break
    else:
        print("ChatBuddy: I'm not sure how to respond to that. 🤔")
