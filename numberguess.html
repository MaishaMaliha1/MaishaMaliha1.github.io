🎯 Let's Learn Number Guessing Game in Python!

import random

secret = random.randint(1, 10)
tries = 3

while tries > 0:
    guess = int(input(f"You have {tries} tries. Guess: "))
    if guess == secret:
        print("🎉 You guessed it!")
        break
    elif guess < secret:
        print("Too low!")
    else:
        print("Too high!")
    tries -= 1

if guess != secret:
    print("Game over. The number was", secret)
